#include <SFML/Graphics.hpp>
#include <iostream>
int main() 
{
    sf::RenderWindow mainwindow(sf::VideoMode(1920,1080),L"千恋万花",sf::Style::Default);   
    //想适应显示器大小，第一项可以sf::VideoMode::getDesktopMode()
    //最后一项sf::Style::Default是默认值，包含sf::Style::Titlebar （有标题栏）|sf::Style::Close（有关闭按钮）|sf::Style::Resize（可以改变大小）
    //无边框时候可以sf::Style::None
    //全屏无边框的时候可以sf::Style::Fullscreen
    //mainwindow.create(sf::VideoMode(800,600),L"魔女的夜宴",sf::Style::None;)   //用新参数重新创建窗口
    //mainwindow.setMouseCursorVisible(false);   //鼠标在窗口里面的时候不显示鼠标
    //mainwindow.setMouseCursorGrabbed(true);    //鼠标不能离开窗口
    mainwindow.setFramerateLimit(30);    //设置帧率
    //mainwindow.setPosition({500,200});    //设置窗口最左上角位置

    while(mainwindow.isOpen()){
        sf::Event event;
        while(mainwindow.pollEvent(event)){     //检测是否关闭窗口，并实现关闭
            if (event.type == sf::Event::Closed){
                mainwindow.close();
            }
        }

        std::cout << mainwindow.hasFocus() << '\n';      //检测焦点在不在窗口，如果在就输出1，反之输出0

        mainwindow.clear();    //清屏
        mainwindow.display();    //重新绘制，这两步可以直接生成每一帧完整的画面，而不是逐步绘制
    }
}
