#include <SFML/Graphics.hpp>
#include <iostream>
int main() 
{
    sf::RenderWindow mainwindow(sf::VideoMode(1920,1080),L"千恋万花",sf::Style::Default); 
    mainwindow.setFramerateLimit(30);    //设置帧率


    while(mainwindow.isOpen()){
        sf::Event event;
        while(mainwindow.pollEvent(event)){     //检测操作，并实现
            if (event.type == sf::Event::Closed){  //检测并实现关闭
                mainwindow.close();
            }

            if (event.type == sf::Event::Resized) {   //检测屏幕大小是否修改并输出修改后的值
                std::cout << "Size: " << event.size.width << 'x' << event.size.height << '\n';
            }       
            if (event.type == sf::Event::LostFocus) {   //检测是否失焦
                std::cout<<"Focus Lost"<<'\n';
            }
            if (event.type == sf::Event::KeyPressed) {  //检测按键按下
                if (event.key.code == sf::Keyboard::W){  //若想检测几个按键一起，sf::Keyboard::W && event.key.control
                    std::cout<<"w"<<'\n';
                }
            }
            if (event.type == sf::Event::KeyReleased) {   //检测按键松开
                if (event.key.code == sf::Keyboard::W){
                    std::cout<<"w"<<'\n';
                }
            }
            if (event.type == sf::Event::MouseButtonPressed){    //检测鼠标按键，输出按键位置
                if (event.mouseButton.button == sf::Mouse::Left){
                    std::cout<<"l "<<event.mouseButton.x<<", "<<event.mouseButton.y<<'\n';
                }
            }
            if (event.type == sf::Event::MouseWheelMoved) {  //检测鼠标滚轮是否滚动，向上输出1，向下输出-1
                    std::cout<< "Wheel delta: " << event.mouseWheel.delta << '\n';
                    std::cout<< "Wheel at " << event.mouseWheel.x << ", " << event.mouseWheel.y<<"\n";  //滚轮位置
            }

        }

        std::cout << mainwindow.hasFocus() << '\n';      //检测焦点在不在窗口，如果在就输出1，反之输出0

        mainwindow.clear();    //清屏
        mainwindow.display();    //重新绘制，这两步可以直接生成每一帧完整的画面，而不是逐步绘制
    }
}
